import os
import numpy as np
from pathlib import Path

import torch
import torch.nn as nn
import torchvision.models as models
import torchvision.transforms as transforms
from PIL import Image

NUM_CLASSES = 8
MODEL_FILENAME = "best_model.pt"
IMAGE_EXTENSIONS = {".jpg", ".jpeg", ".png", ".bmp", ".webp"}


class LetterboxToSquare:
    def __init__(self, size=224, fill=(114, 114, 114)):
        self.size = size
        self.fill = fill

    def __call__(self, img):
        from PIL import Image as PILImage
        w, h = img.size
        scale = self.size / max(w, h)
        new_w = max(1, int(round(w * scale)))
        new_h = max(1, int(round(h * scale)))
        img = img.resize((new_w, new_h), PILImage.Resampling.BILINEAR)
        canvas = PILImage.new("RGB", (self.size, self.size), self.fill)
        canvas.paste(img, ((self.size - new_w) // 2, (self.size - new_h) // 2))
        return canvas


VAL_TRANSFORM = transforms.Compose([
    LetterboxToSquare(224),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
])


def get_device():
    if torch.cuda.is_available():
        return torch.device("cuda")
    return torch.device("cpu")


def build_model(num_classes=NUM_CLASSES):
    model = models.resnet50(weights=None)
    model.fc = nn.Sequential(
        nn.Linear(2048, 1024),
        nn.BatchNorm1d(1024),
        nn.ReLU(),
        nn.Dropout(0.5),
        nn.Linear(1024, 512),
        nn.BatchNorm1d(512),
        nn.ReLU(),
        nn.Dropout(0.4),
        nn.Linear(512, num_classes),
    )
    return model


def load_trained_model(model_path, device):
    model = build_model()
    try:
        state = torch.load(model_path, map_location=device, weights_only=False)
    except TypeError:
        state = torch.load(model_path, map_location=device)
    for k in list(state.keys()):
        state[k] = state[k].float()
    model.load_state_dict(state)
    model.to(device)
    model.eval()
    return model


def collect_test_images(test_dir):
    test_path = Path(test_dir)
    items = []
    for folder_id in range(1, 9):
        sub = test_path / str(folder_id)
        if not sub.exists():
            continue
        for img_path in sorted(sub.iterdir()):
            if img_path.is_file() and img_path.suffix.lower() in IMAGE_EXTENSIONS:
                items.append(img_path)
    return items


def Predict(file_path):

    device = get_device()
    script_dir = Path(__file__).parent if "__file__" in dir() else Path.cwd()
    model_path = None
    for candidate in [script_dir / MODEL_FILENAME, Path(MODEL_FILENAME), Path.cwd() / MODEL_FILENAME, Path("models") / MODEL_FILENAME]:
        if candidate.is_file():
            model_path = candidate
            break
    if model_path is None:
        raise FileNotFoundError(f"{MODEL_FILENAME} bulunamadi.")

    model = load_trained_model(str(model_path), device)

    image_paths = collect_test_images(file_path)
    total = len(image_paths)
    print(f"Toplam {total} gorsel bulundu. Tahmin basliyor...")

    image_names = []
    preds = []

    for idx, img_path in enumerate(image_paths, 1):
        img = Image.open(img_path).convert("RGB")
        base = VAL_TRANSFORM(img).unsqueeze(0).to(device)

        with torch.no_grad():
            variants = [base, torch.flip(base, dims=[3])]
            _, _, h, w = base.shape
            cp = int(w * 0.08)
            if cp > 0 and w > cp * 2:
                variants.append(torch.nn.functional.interpolate(
                    base[:, :, :, cp:], size=(h, w), mode='bilinear', align_corners=False))
                variants.append(torch.nn.functional.interpolate(
                    base[:, :, :, :-cp], size=(h, w), mode='bilinear', align_corners=False))
            tcp = int(h * 0.06)
            if tcp > 0 and h > tcp * 2:
                variants.append(torch.nn.functional.interpolate(
                    base[:, :, tcp:, :], size=(h, w), mode='bilinear', align_corners=False))

            probs = np.mean([
                torch.softmax(model(v), dim=1)[0].cpu().numpy() for v in variants
            ], axis=0)
            pred_idx = int(np.argmax(probs))

        image_names.append(img_path.name)
        preds.append(pred_idx + 1)

        if idx % 50 == 0 or idx == total:
            print(f"  [{idx}/{total}] islendi...")

    filename = f"Preds.txt"
    f = open(filename, "w", encoding="utf-8")

    for i in range(len(preds)):

        line = (
            f"{image_names[i]} | Pred: {preds[i]} "
        )

        print(line)

        f.write(line + "\n")

    f.close()
